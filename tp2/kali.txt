# connecter kali
importer dans gns
ajouter une deuxième interface (gns + vb)
connecter au switch
- configurer le port en vlan 20 :
conf t
interface Ethernet0/3
switchport mode access
switchport access vlan 20
no shut
end
write memory

- ajouter l'interface networkmanager :
sudo nmcli connection add type ethernet ifname eth0 con-name "gns3" ipv4.method auto
sudo nmcli connection up gns3


# rog dhcp
- sudo apt install dnsmasq -y
- sudo nano /etc/dnsmasq.conf :
interface=enp0s3
dhcp-authoritative
port=0

dhcp-range=10.1.20.200,10.1.20.250,255.255.255.0,12h
dhcp-option=option:router,10.1.20.254
dhcp-option=option:dns-server,1.1.1.1

- sudo systemctl restart dnsmasq

tout redemarrer
